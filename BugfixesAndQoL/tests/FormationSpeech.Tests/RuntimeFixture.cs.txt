  static void Check(bool b,string m) { if(!b)throw new Exception(m); }
  public static void Run() {
   foreach(int button in new[]{0,1}) {
    var r=new FormationRuntime();var n=r.Setup();try {
     r.drag.CommandButton=button;r.EngineRunHook(false);
     Check(n.Captured==1&&n.Emitted==1&&n.Last.Response==6,"Captured pre-dispatch near voice");
     Check(r.moves==1&&r.originalRuns==1&&EditorDirector.instance.Consumed,"Exactly one move and consumed release");
     r.EngineRunHook(false);Check(n.Emitted==1&&r.moves==1&&r.originalRuns==2,"Next run cannot repeat accepted order");
     Check(r.markerRenderer.failures==0,"Marker renderer remains active");
    } finally { r.Free(); }
   }
   foreach(string scenario in new[]{"rejected","capture-error","audio-error","busy","no-buffer","map-change","disabled","remote-player","reused-group","native-outside-run","engineer","embedded","excluded","far","successive"}) {
    var r=new FormationRuntime();var n=r.Setup();try {
     if(scenario=="rejected")r.accept=false;
     if(scenario=="capture-error")n.ThrowCapture=true;
     if(scenario=="audio-error")n.ThrowAudio=true;
     if(scenario=="busy")n.Busy=true;
     if(scenario=="excluded")n.Voice=eChimps.CHIMP_TYPE_CATAPULT;
     if(scenario=="engineer") { APIShared.UnitAccess.Units->r_UnitChimp=eChimps.CHIMP_TYPE_ENGINEER;APIShared.UnitAccess.Units->N000000C4=1; }
     if(scenario=="embedded")GameTileManagerAPI.Instance.TileManager.LogicGrid[0]=0x40000000;
     if(scenario=="far")r.drag.Target.NativeX=73;
     bool delayed=scenario=="no-buffer"||scenario=="map-change"||scenario=="disabled"||scenario=="remote-player"||scenario=="reused-group"||scenario=="native-outside-run";
     if(delayed)r.buffer=false;
     r.EngineRunHook(false);
     Check(r.moves==1&&r.originalRuns==1&&!r.localFailed,scenario+": failure isolated from engine/movement");
     if(scenario=="rejected")Check(!EditorDirector.instance.Consumed&&n.Emitted==0,"Rejected input left to Vanilla");
     else Check(EditorDirector.instance.Consumed,"Accepted input remains consumed");
     if(delayed) {
      Check(n.Emitted==0,"No-buffer retained");
      if(scenario=="map-change")r.commandRuntime.mapEpoch++;
      if(scenario=="disabled")r.enabled=false;
      if(scenario=="remote-player")GamePlayerManagerAPI.Instance.Player=2;
      if(scenario=="reused-group")GameTribeManagerAPI.Instance.Tribe->r_GlobalId++;
      if(scenario=="native-outside-run") { r.markerRenderer.Render();Check(n.Emitted==0,"Unrelated render cannot consume feedback"); }
      r.buffer=true;r.EngineRunHook(false);
      Check(n.Emitted==((scenario=="no-buffer"||scenario=="native-outside-run")?1:0),scenario+": context gate");
     }
     if(scenario=="busy") { n.Busy=false;r.EngineRunHook(false);Check(n.Emitted==0,"Suppression not replayed"); }
     if(scenario=="audio-error")Check(r.speechNative==null&&r.markerRenderer.failures==0,"Audio-only disable");
     if(scenario=="far")Check(n.Last.Response==10&&n.Last.TargetTileId==0,"Far voice");
     if(scenario=="successive") { r.drag=new ActiveDrag();r.EngineRunHook(false);Check(r.moves==2&&n.Emitted==2,"Two gestures, two responses"); }
     if(scenario=="engineer"||scenario=="embedded"||scenario=="excluded"||scenario=="capture-error")Check(n.Emitted==0,scenario+": exclusion");
    } finally { r.Free(); }
   }
   // Received simulation orders never pass the local release handoff.
   var remote=new FormationRuntime();var rn=remote.Setup();try { remote.TryDispatch(new(),out _);remote.drag=null;remote.EngineRunHook(false);Check(rn.Emitted==0,"Remote execution has no local feedback"); } finally {remote.Free();}
   var locked=new FormationRuntime();var ln=locked.Setup();try {
    using var entered=new ManualResetEventSlim();using var release=new ManualResetEventSlim();
    var worker=new Thread(()=> { lock(locked.speechEngineLock){entered.Set();release.Wait();} });worker.Start();entered.Wait();
    try { Check(locked.CaptureSpeech(locked.drag)==null&&ln.Captured==0,"Busy engine skips native query without blocking"); }
    finally { release.Set();worker.Join(); }
    locked.speechQueue.Enqueue(locked.CaptureSpeech(locked.drag));locked.nativeFeedbackRunActive=true;
    locked.markerRenderer.Render();Check(ln.Emitted==0,"Unprotected native callback cannot emit");
    lock(locked.speechEngineLock)locked.markerRenderer.Render();Check(ln.Emitted==1,"Protected native callback emits once");
   } finally { locked.Free(); }
   var unavailable=new FormationRuntime();unavailable.Setup();try {
    unavailable.InitializeSpeech();Check(unavailable.speechNative==null&&!unavailable.localFailed,"Adapter init failure is audio-only");
    unavailable.EngineRunHook(false);Check(unavailable.moves==1&&unavailable.originalRuns==1&&unavailable.markerRenderer.failures==0,"Formation survives unavailable adapter");
   } finally { unavailable.Free(); }
   var renderer=new Renderer();renderer.NativeFrameCompleted=()=>throw new Exception("unexpected callback failure");
   renderer.Render();Check(renderer.resets==1&&renderer.failures==0,"Unexpected feedback exception cannot repeat reset or disable markers");
  }
