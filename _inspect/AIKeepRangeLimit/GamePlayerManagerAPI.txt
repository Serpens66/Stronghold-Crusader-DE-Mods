	/// <summary>
	/// A set of common, reusable predicates for convenience.
	/// </summary>
	public static class PlayerResourcePredicates
	{
		/// <summary>A predicate that matches player resources that is currently active.</summary>
		public static readonly RefPredicate<GamePlayerResources> IsPaused = delegate(in GamePlayerResources playerResources)
		{
			return playerResources.r_IsPaused == 1;
		};

		/// <summary>A predicate that matches player resources that has been marked for deletion.</summary>
		public static readonly RefPredicate<GamePlayerResources> IsNotPaused = delegate(in GamePlayerResources playerResources)
		{
			return playerResources.r_IsPaused == 0;
		};

		/// <summary>A predicate that matches player resources that are still alive</summary>
		public static readonly RefPredicate<GamePlayerResources> IsAlive = delegate(in GamePlayerResources playerResources)
		{
			return playerResources.r_IsPaused == 0 && playerResources.r_WinLossState == WinLossState.None && playerResources.r_LordUnitId != 0;
		};

		/// <summary>A predicate that matches any player resources, used as a default for generic queries.</summary>
		public static readonly RefPredicate<GamePlayerResources> Any = delegate
		{
			return true;
		};
	}

	/// <summary>
	/// Attempts to retrieve a direct, raw pointer to a player's resource data structure by their ID.
	/// </summary>
	/// <param name="playerId">The ID of the player (1-8).</param>
	/// <param name="resources">When this method returns, contains a pointer to the player's resources if found; otherwise, null.</param>
	/// <returns><c>true</c> if the player ID was valid and the pointer was retrieved; otherwise, <c>false</c>.</returns>
	[MethodImpl(MethodImplOptions.AggressiveInlining)]
	public unsafe bool TryGetPlayerResourcesById(int playerId, out GamePlayerResources* resources)
	{
		resources = null;
		if (!IsPlayerIdValid(playerId))
		{
			LogHelper.Error($"Tried to access player resources index that was out of range: [{playerId}/{8}]", "TryGetPlayerResourcesById", "C:\\Users\\Serpens66\\AppData\\Local\\Temp\\shcdese-local-build-19079-6972\\source\\src\\SHCDESE.BepInEx\\API\\GamePlayerManagerAPI.cs");
			return false;
		}
		if (_playerResources._array == null)
		{
			return false;
		}
		resources = _playerResources._array + (playerId - 1);
		return true;
	}

	/// <summary>
	/// Attempts to retrieve a safe, wrapped pointer to a player's resource data structure by their ID.
	/// </summary>
	/// <param name="playerId">The ID of the player (1-8).</param>
	/// <param name="resources">When this method returns, contains a <see cref="T:RedBird.Core.Memory.NativePointer`1" /> wrapping the object if found; otherwise, an invalid pointer.</param>
	/// <returns><c>true</c> if the player ID was valid; otherwise, <c>false</c>.</returns>
	[MethodImpl(MethodImplOptions.AggressiveInlining)]
	public unsafe bool TryGetPlayerResourcesByIdEx(int playerId, out NativePointer<GamePlayerResources> resources)
	{
		GamePlayerResources* resources2;
		bool result = TryGetPlayerResourcesById(playerId, out resources2);
		resources = new NativePointer<GamePlayerResources>(resources2);
		return result;
	}

	/// <summary>Returns the AI recruitment FSM mode stored for a player.</summary>
	[LuaApiExport("GetAIRecruitmentMode")]
	public unsafe AIRecruitmentMode GetAIRecruitmentMode(int playerId)
	{
		GamePlayerResources* resources;
		return TryGetPlayerResourcesById(playerId, out resources) ? resources->r_AICombatRecruitmentMode : AIRecruitmentMode.Defender;
	}

	/// <summary>Writes the player's AI recruitment FSM mode.</summary>
	[LuaApiExport("SetAIRecruitmentMode")]
	public unsafe bool SetAIRecruitmentMode(int playerId, AIRecruitmentMode mode)
	{
		if (mode > AIRecruitmentMode.Siege || !TryGetPlayerResourcesById(playerId, out var resources))
		{
			return false;
		}
		resources->r_AICombatRecruitmentMode = mode;
		return true;
	}

	/// <summary>Returns the live ten-slot defensive-position tile table for one native class.</summary>
	public unsafe Span<uint> GetAIDefensivePositionTileIds(int playerId, AIDefensivePositionClass positionClass)
	{
		if (!IsValidAIDefensivePositionClass(positionClass) || !TryGetPlayerResourcesById(playerId, out var resources))
		{
			return Span<uint>.Empty;
		}
		uint* ptr = resources->r_AIDefensivePositionTileIdsByClass.r_TileIds;
		return new Span<uint>(ptr + (int)positionClass * 10, 10);
	}

	/// <summary>Returns the live 30-entry defensive-position count table for a player.</summary>
	public unsafe Span<int> GetAIDefensivePositionCounts(int playerId)
	{
		if (!TryGetPlayerResourcesById(playerId, out var resources))
		{
			return Span<int>.Empty;
		}
		return new Span<int>(resources->r_AIDefensivePositionCountsByClass.r_Counts, 30);
	}

	[LuaApiExport("GetAIDefensivePositionCount")]
	public int GetAIDefensivePositionCount(int playerId, AIDefensivePositionClass positionClass)
	{
		if (!IsValidAIDefensivePositionClass(positionClass))
		{
			return 0;
		}
		Span<int> aIDefensivePositionCounts = GetAIDefensivePositionCounts(playerId);
		if (aIDefensivePositionCounts.IsEmpty)
		{
			return 0;
		}
		return Math.Min(Math.Max(aIDefensivePositionCounts[(int)positionClass], 0), 10);
	}

	public bool TryGetAIDefensivePositionTileId(int playerId, AIDefensivePositionClass positionClass, int positionIndex, out uint tileId)
	{
		tileId = 0u;
		int aIDefensivePositionCount = GetAIDefensivePositionCount(playerId, positionClass);
		if ((uint)positionIndex >= (uint)aIDefensivePositionCount)
		{
			return false;
		}
		Span<uint> aIDefensivePositionTileIds = GetAIDefensivePositionTileIds(playerId, positionClass);
		if (aIDefensivePositionTileIds.IsEmpty)
		{
			return false;
		}
		tileId = aIDefensivePositionTileIds[positionIndex];
		return true;
	}

	/// <summary>
	/// Writes one raw defensive-position slot. The caller is responsible for keeping the class count coherent.
	/// </summary>
	public bool SetAIDefensivePositionTileIdUnsafe(int playerId, AIDefensivePositionClass positionClass, int positionIndex, uint tileId)
	{
		if ((uint)positionIndex >= 10u)
		{
			return false;
		}
		Span<uint> aIDefensivePositionTileIds = GetAIDefensivePositionTileIds(playerId, positionClass);
		if (aIDefensivePositionTileIds.IsEmpty)
		{
			return false;
		}
		aIDefensivePositionTileIds[positionIndex] = tileId;
		return true;
	}

	/// <summary>Writes the active position count for one defensive class.</summary>

	{
		if (!IsValidAIDefensivePositionClass(positionClass) || (uint)count > 10u)

			return false;

		Span<int> aIDefensivePositionCounts = GetAIDefensivePositionCounts(playerId);
		if (aIDefensivePositionCounts.IsEmpty)
		{
			return false;
		}
		aIDefensivePositionCounts[(int)positionClass] = count;
		return true;
	}

	/// <summary>
	/// Sets the camera raw position directly in Unity World Space.
	/// </summary>
	/// <param name="position">The target Unity World Space coordinate. This is NOT a game tile coordinate.</param>
	/// <remarks>
	/// This function operates on the engine internal camera coordinate system. It does not account for isometric projection, map rotation, or terrain height. 
	/// Using game tile coordinates (e.g., 400, 400) with this function will result in unpredictable camera movement as the coordinates will be misinterpreted.
	/// It is preserved for specific low-level use cases or for restoring a previously saved raw camera position. 
	/// For general use, call <see cref="M:SHCDESE.API.GamePlayerManagerAPI.SetScreenCenterToTilePosition(System.Numerics.Vector2)" /> instead.
	/// </remarks>
	[LuaApiExport("SetScreenCenterPosition")]
	public void SetScreenCenterPosition(Vector2 position)
	{
		UnityMainThreadDispatcher.Instance.Enqueue(delegate
		{
			CameraControls2D.instance.setCameraPos(position.X, position.Y);
			GameMap.instance.PreCalcScreenCentre();
		});
	}

	/// <summary>
	/// Moves the camera to center the view on a specific local game tile coordinate. This is the recommended function for camera positioning.
	/// </summary>
	/// <param name="position">A Vector2 containing the target local tile coordinates (e.g., X=400, Y=400 on a 400x400 map).</param>
	/// <remarks>
	/// This function acts as a translator, converting human-readable grid coordinates into the complex world coordinates required by the camera.
	/// It should be used instead of SetScreenCenterPosition, which expects raw camera coordinates.
	/// The conversion process is as follows:
	/// <para> 1. Takes a Local Tile Coordinate (the game logical grid). </para>
	/// <para> 2. Converts it to an internal Isometric Tilemap Coordinate, accounting for map rotation. </para>
	/// <para> 3. Calculates the base Unity World Position for that tile. </para>
	/// <para> 4. Adjusts the World Position Y-axis based on the tile terrain height. </para>
	/// <para> 5. Passes the final, calculated World Position to the camera controller. </para>
	/// </remarks>
	[LuaApiExport("SetScreenCenterToTilePosition")]
	public void SetScreenCenterToTilePosition(Vector2 position)
	{
		Vector2 cameraWorldPosition = CoordinateConverter.ConvertLocalTileToCameraWorld(position);
		UnityMainThreadDispatcher.Instance.Enqueue(delegate
		{
			if ((Object)(object)CameraControls2D.instance != (Object)null)
			{
				CameraControls2D.instance.setCameraPos(cameraWorldPosition.X, cameraWorldPosition.Y);
			}
			if ((Object)(object)GameMap.instance != (Object)null)
			{
				GameMap.instance.PreCalcScreenCentre();
				GameMap.instance.ignoreNextCachedBounds = false;
			}
		});
	}

	/// </summary>

	{
		int localPlayerId = GameNetworkAPI.GetLocalPlayerId();
		if (localPlayerId > 8 || localPlayerId < 1)

			return Array.Empty<SelectedUnitInfo>();
		}
		int selectedChimpsCount = GetSelectedChimpsCount(localPlayerId);
		List<SelectedUnitInfo> list = new List<SelectedUnitInfo>(selectedChimpsCount);
		fixed (int* selectedChimps = EngineInterface.selectedChimps)
		{
			Span<SelectedUnitInfo> span = new Span<SelectedUnitInfo>(selectedChimps, selectedChimpsCount);
			for (int i = 0; i < selectedChimpsCount; i++)

				list.Add(span[i]);
			}
		}
		return list.ToArray();
	}

	[Obsolete("Do not use.")]
	public unsafe bool TryRemoveGoodEx(int playerId, eGoods good, int toRemoveAmount, bool onlyRemoveIfHasEnough)
	{
		if (toRemoveAmount == 0)
		{
			return true;
		}
		if (!TryGetPlayerResourcesById(playerId, out var _))

			LogHelper.Warning($"Could not find player resource by id: {playerId}", "TryRemoveGoodEx", "C:\\Users\\Serpens66\\AppData\\Local\\Temp\\shcdese-local-build-19079-6972\\source\\src\\SHCDESE.BepInEx\\API\\GamePlayerManagerAPI.cs");

		GameStructQuery<GameBuilding> gameStructQuery = GameBuildingManagerAPI.Instance.QueryBuildings().Where(GameBuildingManagerAPI.BuildingPredicates.IsStorage).Where(GameBuildingManagerAPI.BuildingPredicates.IsOwnedByPlayer(playerId))
			.Where(GameBuildingManagerAPI.BuildingPredicates.ContainsGoodInAnyStorage(good));
		long num = 0L;
		List<NativePointer<GameBuilding>> list = new List<NativePointer<GameBuilding>>();
		GameStructQuery<GameBuilding>.Enumerator enumerator = gameStructQuery.GetEnumerator();
		while (enumerator.MoveNext())
		{
			ref GameBuilding current = ref enumerator.Current;
			num += current.GetLocalGoodsAmount(good);
			list.Add((GameBuilding*)Unsafe.AsPointer(ref Unsafe.AsRef(in current)));
		}
		if (onlyRemoveIfHasEnough && num < toRemoveAmount)
		{
			return false;
		}
		long num2 = Math.Min(toRemoveAmount, num);
		if (num2 == 0)
		{
			return true;
		}
		long num3 = num2;
		foreach (NativePointer<GameBuilding> item in list)
		{
			GameBuilding* source = item;
			if (num3 <= 0)
			{
				break;
			}
			ref GameBuilding self = ref Unsafe.AsRef<GameBuilding>(source);
			int localGoodsAmount = self.GetLocalGoodsAmount(good);
			long num4 = Math.Min(num3, localGoodsAmount);
			self.RemoveLocalGoodsAmount(good, (int)(-num4));
			self.UpdateLocalGoodsResourceVisuals();
			num3 -= num4;
		}
		return true;
	}
